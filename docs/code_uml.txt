** CLASSES **

@startuml

enum Role {
  CLIENT
  COLLABORATEUR
  ADMIN
}

enum BookingStatus {
  PAYE
  ANNULE
  EN ATTENTE DE PAIEMENT
}

class User {
  + id : Integer
  + username : String
  + email : String
  + passwordHash : String
  + role : Role
  + subscriptionId : Integer
  + register()
  + login()
}

class Subscription {
  + id : Integer
  + name : String
  + discountPercentage : Float
  + conditions : String
}

class Station {
  + id : Integer
  + codeGareSNCF : String
  + name : String
  + city : String
}

class StationAdjacency {
  + id : Integer
  + station1Id : Integer
  + station2Id : Integer
  + railType : String
}

class OperatingLine {
  + id : Integer
  + code : String
  + departureStationId : Integer
  + arrivalStationId : Integer
  + estimatedDurationMinutes : Integer
}

class Trip {
  + id : Integer
  + lineId : Integer
  + departureTime : DateTime
  + arrivalTime : DateTime
  + totalSeats : Integer
  + availableSeats : Integer
  + basePrice : Float
}

class Booking {
  + id : Integer
  + userId : Integer
  + tripId : Integer
  + status : BookingStatus
  + finalPrice : Float
  + createdAt : DateTime
  + cancel()
}

' Relations
User "0..*" -- "0..1" Subscription : souscrit à >
User "1" -- "0..*" Booking : passe >
Booking "0..*" -- "1" Trip : concerne >
Trip "0..*" -- "1" OperatingLine : s'exécute sur >
OperatingLine "1" -- "2" Station : relie >
StationAdjacency "0..*" -- "2" Station : connecte >

@enduml

** SEQUENCE **

@startuml
autonumber
skinparam style strictuml
skinparam sequenceMessageAlign center

actor "Client / Collab" as Client
participant "TripController" as Ctrl
participant "TripService" as Service
participant "SNCFClientService" as SNCFService
participant "API SNCF Externe" as API_SNCF
participant "StationRepository" as StationDAO
participant "TripRepository" as TripDAO
database "PostgreSQL" as BDD

Client -> Ctrl : GET /api/trips?dep=Rennes&arr=Paris
activate Ctrl

  Ctrl -> Service : searchTrips("Rennes", "Paris")
  activate Service

    ' 1. Vérification / Récupération des gares en BDD
    Service -> StationDAO : findByName("Rennes"), findByName("Paris")
    activate StationDAO
    StationDAO -> BDD : SELECT * FROM stations WHERE name IN (...)
    activate BDD
    BDD --> StationDAO : Données gares (codes UIC)
    deactivate BDD
    StationDAO --> Service : Objets Station (Rennes, Paris)
    deactivate StationDAO

    ' 2. Appel à l'API SNCF pour le calcul du temps de trajet
    Service -> SNCFService : getJourneyDuration(uicDep, uicArr)
    activate SNCFService
      SNCFService -> API_SNCF : GET /journeys (UIC départ, UIC arrivée)
      activate API_SNCF
      API_SNCF --> SNCFService : 200 OK (Durée calculée : 45 min)
      deactivate API_SNCF
    SNCFService --> Service : 45 minutes
    deactivate SNCFService

    ' 3. Recherche des trajets planifiés correspondants
    Service -> TripDAO : findAvailableTrips(depId, arrId)
    activate TripDAO
      TripDAO -> BDD : SELECT * FROM trips WHERE line_id = ...
      activate BDD
      BDD --> TripDAO : Liste des trajets bruts
      deactivate BDD
    TripDAO --> Service : Liste d'objets Trip
    deactivate TripDAO

    ' 4. Application de la logique métier
    Service -> Service : Enrichir trajets avec durée SNCF\net décomposition tarifaire (Infra/Énergie)

  Service --> Ctrl : Liste des trajets enrichis (DTO)
  deactivate Service

Ctrl --> Client : 200 OK (JSON avec trajets, prix décomposé, durée)
deactivate Ctrl

@enduml
